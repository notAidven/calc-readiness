The fire's four states

Output: three.js code

Four hearths in a row, left to right: proposed, declared and never caught, burning, burnt out. Every one is the same object — the nine-stone ring, three laid logs, and for the two that are alight a seven-sided ember bed under seven lathed tongues. Batch 1 built that object and it is not rebuilt here; what this asset settles is what changes between the four, and the answer is that each state changes the hearth and the ground under it together. Three of the four re-tone the tile the hearth stands on, and that is what stops the row reading as one hearth with a lighting bug.

import * as THREE from 'three';
import { WINTER_SCENE_TONES as W, WINTER_GROUND_TONES as G } from './winterTones';
import { ALL_EXPOSED, buildCellTerrain, flatWinter } from './winterScene';
import { cellCenter } from '../worldLayout';
import { buildHearth, buildSmokeColumn, FUEL_NAME } from './winterFire';
import { buildGhostHearth, buildFireGhostPatch, WARDED_TERRAIN } from './winterFireGhost';

export const FIRE_STATES = ['proposed', 'unlit', 'burning', 'spent'] as const;

// ── THE GROUND IS HALF OF EVERY STATE ────────────────────────────────
// Held ground closing over again is its own terrain spec and NOT the ordinary frost
// tile: it keeps the warded patch's dish, segment count and feathered rim, so panel 4
// reads as THE SAME PATCH refrozen rather than as a new cell that arrived from
// somewhere. Only the colour moves — warded ochre out, frost in.
export const REFROZEN_TERRAIN = {
  color: G.frost, dish: 0.12, segments: 4, edge: 'feather',
  values: [0.72, 0.84, 0.93, 1.0],
};

const GROUND_FOR = {
  proposed: 'ghost-warded',   // the real warded terrain, translucent, per tier
  unlit:    null,             // snow, unbroken and unthawed. The absence IS the state.
  burning:  WARDED_TERRAIN,   // opaque ochre, the one definition
  spent:    REFROZEN_TERRAIN, // the ochre has gone back under ice
};

// ── THE SPENT HEARTH ─────────────────────────────────────────────────
// Two tones, both STONE and STONE_DARK lerped 0.55 toward CHAR. Chosen at BUILD time,
// not tinted at runtime: nothing on this island tints a mesh it did not make, and this
// hearth is ours and procedural, so a spent fire is a different material set rather
// than the same material set with a filter over it.
const STONE_SPENT      = 0x6a635a;
const STONE_DARK_SPENT = 0x554f45;

export function buildSpentHearth(radius: number): THREE.Group {
  const g = buildHearth(radius, false);        // the UNLIT build: no ember bed, no flame
  g.name = 'winter-hearth-spent';
  let stone = 0;
  for (const child of g.children) {
    const mesh = child as THREE.Mesh;
    if (mesh.name === FUEL_NAME) {
      mesh.material = flatWinter({ color: W.CHAR });
      continue;
    }
    mesh.material = flatWinter({ color: stone++ % 2 ? STONE_SPENT : STONE_DARK_SPENT });
  }
  // The stack has collapsed into its own ash: every log drops and rolls a little.
  g.children.filter(m => m.name === FUEL_NAME).forEach((log, i) => {
    log.position.y = radius * (0.13 + i * 0.05);   // was 0.19 + i * 0.13
    log.rotation.z += (i % 2 ? 1 : -1) * 0.14;
  });
  return g;
}

// ── THE WISP ─────────────────────────────────────────────────────────
// A spent fire gets three puffs, low and thin, and they do not recycle. Built off the
// real column so there is still exactly one smoke geometry in the codebase.
export function buildWisp(seed: number, windDeg: number): THREE.Group {
  const g = buildSmokeColumn(seed, windDeg);
  g.name = 'winter-smoke-wisp';
  for (const extra of g.children.slice(3)) g.remove(extra);
  g.children.forEach((puff, i) => {
    const mesh = puff as THREE.Mesh;
    puff.scale.multiplyScalar(0.42);
    mesh.material = (mesh.material as THREE.Material).clone();
    (mesh.material as THREE.MeshStandardMaterial).opacity = 0.30 - i * 0.07;
  });
  g.userData = { ...g.userData, still: true };   // updateFire() skips a still column
  return g;
}

// ── THE FOUR STATES, AS ONE CALL ─────────────────────────────────────
export function buildFireState(
  state: (typeof FIRE_STATES)[number],
  { radius = 1.0, seed = 1, sc-camel-wind-deg = 34, col = 0, row = 0, sc-camel-top-y = 0.2, depth = 0.34 } = {},
): THREE.Group {
  const group = new THREE.Group();
  group.name = `winter-fire-${state}`;

  // GROUND FIRST, so a state that has none is a hole in the code and not an oversight.
  // `buildCellTerrain` takes WORLD coordinates, as it does at every existing call site;
  // `buildFireGhostPatch` takes the cell and resolves them itself.
  const { x, z } = cellCenter(col, row);
  const spec = GROUND_FOR[state];
  if (spec === 'ghost-warded') {
    group.add(buildFireGhostPatch(col, row, 'steady', topY, depth));   // opacity 0.58
  } else if (spec) {
    group.add(buildCellTerrain(spec, x, z, topY, depth, ALL_EXPOSED));
  }

  if (state === 'proposed') {
    // 0.52 opacity, depthWrite off, flame emissive × GHOST_HEAT 0.45, casts nothing.
    // Three channels say "proposal" and NONE of them is a colour: a ghost fire is the
    // fire's own tones, because a proposal in a different hue is a second vocabulary.
    group.add(buildGhostHearth(radius));
  } else if (state === 'unlit') {
    // Stones, three laid logs, the third in WOOD_LIT because it has never been burnt.
    // No bed, no flame, no smoke, no held ground. Snow runs right up to the stones.
    group.add(buildHearth(radius, false));
  } else if (state === 'burning') {
    group.add(buildHearth(radius, true));            // third log CHAR, bed + 7 tongues
    group.add(buildSmokeColumn(seed, windDeg));      // 26 units, lean 0.17 per unit
  } else {
    group.add(buildSpentHearth(radius));
    group.add(buildWisp(seed, windDeg));
  }
  return group;
}
Told apart

Proposed against burning is the hard pair, and it is settled on four channels at once, none of them hue: opacity 0.52, depthWrite: false so the ghost never occludes what stands behind it, the flame's emissive ramp damped to 0.45 of authored, and no shadow on the ground. Put the two side by side and the real one is the solid one. Unlit against a hearth that failed to draw is settled by the fuel: three logs are laid in it, pale, and a ring of stones with wood in it is obviously a built thing doing nothing. Spent against burning changes two things at once on purpose — the stones darken and the ground reverts — because one change alone reads as a lighting difference between two panels.

Animation

updateFire(hearth, column, t) as shipped, for burning only. The proposed hearth breathes on the same call at half amplitude — a ghost held perfectly still beside a live fire reads as a frame drop, not as a proposal. Unlit and spent never animate; the wisp is posed once and left. Reduced motion: call nothing. The flame holds its authored pose, the column stands where it was built, and the still frame is a legible fire — which is the same treatment StarterIsland gives its gulls.
Traded away

Plate 07 gives the proposed hearth a faint translucent shadow. Dropped: a proposal casts nothing and catches nothing, which is a rule the ghost shield is under too, so a ghost with a shadow would be the only inconsistent proposal on the board. The plate also draws the smoke as one tapered ribbon; a ribbon has to face the camera and the camera orbits a full circle, so it stays eleven discrete icosahedra.

Duration lives in the fuel

Output: three.js code

Three burning hearths, identical except for how much fuel is left: three logs, two, one. burn_rounds_by_tier runs 1, 2, 2, 3, so three steps covers the whole range and no number is invented. The asset exists to settle one thing and it is a negative: the third hearth must read as the same fire later, never as a smaller fire.

import { buildHearth, FUEL_NAME, FLAME_NAME } from './winterFire';
import { buildCellTerrain } from './winterScene';
import { WARDED_TERRAIN } from './winterFireGhost';

/** hard-winter-tuning.ts: the whole range a fire can have. Three steps covers it. */
export const BURN_ROUNDS_BY_TIER = [1, 2, 2, 3] as const;
export const FUEL_STEPS = [3, 2, 1] as const;

/**
 * **THE FIRE KEEPS ITS FOOTPRINT AND LOSES ITS HEIGHT, AND THAT IS THE WHOLE ASSET.**
 *
 * Every radius in the hearth is untouched at every step: the stone ring, the ember bed's
 * seven-sided disc, and every tongue's BASE radius. Only Y moves. A fire drawn narrower
 * as it burns down is a fire that covers less ground, which is the one reading this
 * frame exists to prevent — duration is not width.
 *
 * The core tongue falls faster than the skirt (0.38 against 0.48 at one log), because a
 * dying fire loses its peak before it loses its lap: keeping the ratio flat gave three
 * flames that looked like one flame at three zoom levels.
 */
const FUEL_POSE = {
  3: { logs: 3, skirtY: 1.00, coreY: 1.00 },
  2: { logs: 2, skirtY: 0.72, coreY: 0.66 },
  1: { logs: 1, skirtY: 0.48, coreY: 0.38 },
} as const;

export function buildBurningHearth(radius: number, roundsLeft: 1 | 2 | 3): THREE.Group {
  const pose = FUEL_POSE[roundsLeft];
  const g = buildHearth(radius, true);

  // Take logs off the TOP. buildHearth lays them bottom-first and puts CHAR on the
  // third, so the log that has already been in the fire is the last one to go — the
  // stack reads as burning down rather than as three different stacks.
  const logs = g.children.filter((m) => m.name === FUEL_NAME);
  for (const spent of logs.slice(pose.logs)) g.remove(spent);

  const flame = g.getObjectByName(FLAME_NAME)!;
  flame.children.forEach((tongue, i) => {
    // Y ONLY. x and z stay at exactly 1 — see the note above.
    tongue.scale.set(1, i === 0 ? pose.coreY : pose.skirtY, 1);
  });
  // The ember bed is NOT scaled. It is the floor of the fire and its diameter is the
  // clearest statement in the frame that the fire is the same size it always was.
  return g;
}

/**
 * The held ground, and it is the SAME CALL for all three. One spec, one extent, one
 * dish, one rim. Not parameterised by fuel, and there is deliberately no argument here
 * that could be.
 */
export function heldGroundFor(col: number, row: number, topY: number, depth: number) {
  const { x, z } = cellCenter(col, row);
  return buildCellTerrain(WARDED_TERRAIN, x, z, topY, depth);
}

Told apart

Two channels carry the step and both are vertical: the fuel stack's height and the flame's height. Nothing horizontal moves anywhere in the frame. The held patch is byte-identical between the three panels — same size, same feathered rim, same distance from the stones — and the ember bed's seven-sided disc is the fixed reference the eye measures the shrinking flame against. If a reviewer reads the third hearth as covering less ground, the fault will be a radius that got scaled, not a fuel step that was too small.

Animation

updateFire multiplies its per-tongue jitter into the pose rather than replacing it: tongue.scale.y = pose * (1 + sin(t·8.1 + φ)·0.17). Left as an authored scale that the frame loop overwrites, a one-log fire snaps back to full height on the first frame. Reduced motion: the pose alone, which is the whole read — the fuel step is static geometry and loses nothing.
Traded away

A second channel for remaining rounds — a rail readout beside the fuel stack — is owed to the lead and not decided here. The fuel stack may well be enough at the resting camera and is certainly not enough at full zoom-out, but adding a rail is a UI ruling rather than a scene asset, so this frame draws the stack and nothing else. Nothing floats over any hearth in it.

Held ground against the frost's edge

Output: three.js code

The winter draws two kinds of boundary on the ground and they must never be mistaken for each other. The frost's edge is a cell boundary: hard, stepped, corner to corner, countable. The fire's held ground is not another cell: its rim is pulled back off the cell lines and sunk below the snow, so it is soft-edged — and still bounded, because held ground is the one thing in the crisis a player aims. One builder makes both, and the field that differs is edge.

import * as THREE from 'three';
import { GRID_CELL, cellCenter } from '../worldLayout';
import { ALL_EXPOSED, type CellExposure, facedMaterial, flatWinter } from './winterScene';
import { WINTER_GROUND_TONES as G, WINTER_SCENE_TONES as W } from './winterTones';

/** The shared decal ladder. Flat things on the ground sit on 0.05-unit rungs. */
export const GROUND_DECAL_STEP = 0.05;
export const WINTER_COURSE_TOP = 0.20;     // batch 1's snow course, at its crown

// Rung 1 for any ground STATE tile; rung 2 for the leading edge, so the frost line
// always wins the z-fight against the tile it is announcing. Standing geometry (the
// shove ridge) sits ON the ground rather than in the ladder.
const RUNG = { tile: 1, line: 2 };

// ── TWO EDGES, ONE BUILDER ───────────────────────────────────────────
export const FROST_TERRAIN  = { color: G.frost,      dish: 0,    segments: 1, edge: 'step',    values: [0.88, 0.94, 1.0, 1.0] };
export const LINE_TERRAIN   = { color: G.frostLine,  dish: 0,    segments: 1, edge: 'step',    values: [0.92, 1.0, 1.0, 1.0] };
export const WARDED_TERRAIN = { color: G.warded,     dish: 0.12, segments: 4, edge: 'feather', values: [0.70, 0.82, 0.92, 1.0] };

const hash = (x: number, z: number) => {
  const v = Math.sin(x * 12.9898 + z * 78.233) * 43758.5453;
  return v - Math.floor(v);
};

// The shipped signature, unchanged: WORLD x/z, the ladder height, the skirt depth and the
// cell's exposure. Every existing call site passes it this way — `buildFireGhostPatch`
// resolves `cellCenter(col, row)` before it calls in, and so does anything else that has a
// cell rather than a point.
export function buildCellTerrain(
  spec, x: number, z: number, topY: number, depth: number, exposed: CellExposure = ALL_EXPOSED,
) {
  const half = GRID_CELL / 2;
  const g = new THREE.PlaneGeometry(GRID_CELL, GRID_CELL, spec.segments, spec.segments)
    .rotateX(-Math.PI / 2).toNonIndexed();
  const p = g.attributes.position as THREE.BufferAttribute;

  for (let i = 0; i < p.count; i++) {
    const lx = p.getX(i), lz = p.getZ(i);
    const sc-camel-on-rim = Math.abs(Math.abs(lx) - half) < 1e-4 || Math.abs(Math.abs(lz) - half) < 1e-4;

    if (spec.edge === 'step') {
      // SNAP, DO NOT SOFTEN. The tile ends exactly on the cell line, corner to corner,
      // at a flat y. No jitter on the rim, ever: the whole claim of the left half is
      // that a player can COUNT cells along this edge, and one jittered vertex per cell
      // is enough to make a run of eight look like a coastline.
      p.setY(i, 0);
      continue;
    }

    // FEATHER: only the RIM RING moves. Pulled inboard by 0.10–0.26 of the half-cell
    // and sunk into the dish, so the patch ends ragged and BELOW the snow around it.
    // The interior stays flat and full-value, which is what keeps the cells under the
    // patch readable — a fully soft patch is a spotlight pool and this is not one.
    const k = hash(x + lx, z + lz);
    if (onRim) {
      p.setX(i, lx * (1 - 0.10 - 0.16 * k));
      p.setZ(i, lz * (1 - 0.10 - 0.16 * k));
      p.setY(i, -spec.dish * (0.60 + 0.40 * k));
    } else {
      p.setY(i, -spec.dish * 0.35 * k);       // faceted floor, not a flat pan
    }
  }
  g.computeVertexNormals();

  // Per-facet value out of spec.values: the tile is one hue at four values, so the
  // surface facets under flat shading without a second tone entering the palette.
  applyFacetValues(g, spec.color, spec.values);

  const mesh = new THREE.Mesh(g, facedMaterial(g, { color: spec.color, roughness: 0.94 }));
  mesh.position.set(x, topY + GROUND_DECAL_STEP * RUNG.tile, z);
  mesh.receiveShadow = true;
  mesh.castShadow = false;
  // A SKIRT for any dished tile, or the sunken rim shows the snow's underside as a gap.
  // Only on the EXPOSED sides, as shipped: a skirt between two tiles of the same state is
  // interior geometry nobody sees and it z-fights its neighbour's.
  if (spec.dish > 0) mesh.add(buildSkirt(g, depth, spec.color, exposed));
  return mesh;
}

// ── THE FRONT: STANDING GEOMETRY, AND OPAQUE ─────────────────────────
/**
 * A low ridge of shove ice on the row the frost takes next, in frostFront #3C7FA0 —
 * the one dark value in the cold palette.
 *
 * Boxes rather than a tile, because the front is the only ground state that is not
 * flat: it stands, so it survives being seen along the edge rather than across it. And
 * OPAQUE, which is the other half of the lesson — a translucent front over near-white
 * snow composites to a pale tint whatever the arithmetic says.
 */
export function buildShoveRidge(edgeCells: { col: number; row: number; facing: number }[]) {
  const group = new THREE.Group();
  group.name = 'winter-shove-ridge';
  const ice = flatWinter({ color: G.frostFront });
  for (const cell of edgeCells) {
    const { x, z } = cellCenter(cell.col, cell.row);
    const shards = 3;
    for (let i = 0; i < shards; i++) {
      const k = hash(x + i, z - i);
      const h = 0.34 + k * 0.28;                       // 0.34–0.62: a ridge, not a wall
      const shard = new THREE.Mesh(
        new THREE.BoxGeometry(GRID_CELL * (0.26 + k * 0.12), h, GRID_CELL * 0.16), ice);
      shard.position.set(
        x + Math.cos(cell.facing) * GRID_CELL * 0.42 + (i - 1) * GRID_CELL * 0.3,
        WINTER_COURSE_TOP + h / 2,
        z + Math.sin(cell.facing) * GRID_CELL * 0.42);
      shard.rotation.set(0, cell.facing + (k - 0.5) * 0.6, (k - 0.5) * 0.18);
      shard.castShadow = true;
      shard.receiveShadow = true;
      group.add(shard);
    }
  }
  return group;
}

Told apart

Four differences, and they are cumulative rather than alternative. Plan: the frost's rim vertices sit exactly on the cell line; the fire's are pulled inboard by 10–26% of the half-cell. Section: the frost tile is flat at rung 1; the held patch dishes 0.12 below its own top, so its boundary is a lip you can see under, not a colour change. Facets: one segment against four. Company: the frost's edge is announced by a colder line and a standing ridge; held ground is announced by nothing but itself. The two are never both edges of the same tile — a frozen cell is not warded and a warded cell is not frozen — so they never have to be told apart at zero distance.

Traded away

A genuinely soft rim — an alpha falloff at the patch's edge — is what "feathered" would mean in a paint program and it is not available: there is no texture in the renderer and a translucent ochre over snow goes pale. The substitute is geometric feathering, an irregular inboard rim plus a dish, which reads as thawed rather than painted and keeps the cells countable. The other loss is the plate's soft ground shadow under the ridge; shadow colour is not a parameter that exists here, so the ridge takes the one shadow map like everything else.

The five protection states

Output: three.js code

Five buildings in a row: threatened, proposed shield, protected, frosted, ruined. Nothing in the renderer tints a building at runtime — a building's colour is baked into its GLB — so every state here is added geometry, which also means every state survives full zoom-out in silhouette. Batch 1 built the frost crust and the ruin heap and neither is rebuilt here. What this asset adds is the storm roof, its proposal, and the numbers that keep protected and frosted from being the same pale shape.

import * as THREE from 'three';
import { type Footprint, FOOTPRINT_FILL, GRID_CELL } from '../worldLayout';
import { flatWinter } from './winterScene';
import { WINTER_SCENE_TONES as W } from './winterTones';

export const OVERSAIL   = 1.24;              // the eave past the wall — the outline change
const RISE_RATIO = 0.52;                     // of the SHORTER span, so a long lot is not over-roofed
const RISE_CAP   = GRID_CELL * 1.15;         // never more than about a storey
const BATTENS    = 3;                        // countable at full zoom-out
const BERM_RISE  = 1.05, BERM_REACH = 0.8;
export const GHOST_OPACITY = 0.68;           // GhostBuilding uses 0.50; raised off a frame

const sc-camel-wall-span = (f: Footprint) => ({
  x: f.w * GRID_CELL * FOOTPRINT_FILL,
  z: f.d * GRID_CELL * FOOTPRINT_FILL,
});
export const sc-camel-shield-rise = (f: Footprint) =>
  Math.min(RISE_CAP, Math.min(wallSpan(f).x, wallSpan(f).z) * RISE_RATIO);

/**
 * **A GABLE RATHER THAN A HIP, AND THAT IS THE WHOLE SILHOUETTE DECISION.**
 *
 * A hip roof from 49° three-quarters is a trapezoid, which is what most of these
 * buildings already are. A gable adds a hard RIDGE LINE and two raking eaves: three
 * edges the ordinary building does not have, gained at the TOP of the shape, against
 * the sky, which is where an outline reads best.
 */
function gableMesh(width: number, depth: number, rise: number): THREE.Mesh {
  const half = width / 2;
  const section = new THREE.Shape();
  section.moveTo(-half, 0); section.lineTo(half, 0); section.lineTo(0, rise); section.closePath();
  const geometry = new THREE.ExtrudeGeometry(section, { depth, bevelEnabled: false });
  geometry.translate(0, 0, -depth / 2);      // Extrude runs 0→depth; centre it
  const mesh = new THREE.Mesh(geometry, flatWinter({ color: W.WOOD_MID }));
  mesh.name = 'winter-shield-gable';
  mesh.castShadow = mesh.receiveShadow = true;
  return mesh;
}

/** Built with its EAVES at y = 0, so the caller lifts it to the measured roofline. */
export function buildShieldCap(footprint: Footprint): THREE.Group {
  const span = wallSpan(footprint);
  const width = span.x * OVERSAIL, depth = span.z * OVERSAIL;
  const rise = shieldRise(footprint);
  const group = new THREE.Group();
  group.name = 'winter-shield-cap';
  group.add(gableMesh(width, depth, rise));

  // THE RIDGE, standing proud of the apex. A gable's apex is a LINE, and a line at this
  // tilt renders as one pixel row; a squared beam turned corner-up gives the top of the
  // silhouette real thickness. It oversails the gable ends, so it reads from straight on.
  const side = Math.max(rise * 0.17, 0.18);
  const ridge = new THREE.Mesh(
    new THREE.BoxGeometry(side, side, depth * 1.1), flatWinter({ color: W.WOOD_DARK }));
  ridge.position.set(0, rise, 0);
  ridge.rotation.z = Math.PI / 4;
  ridge.castShadow = ridge.receiveShadow = true;
  group.add(ridge);

  // THE LASHINGS. Each batten is TWO bars lying on the two slopes and meeting at the
  // ridge — a single bar spanning the width would float over the middle of a pitched
  // roof, visible as a gap at exactly the tilt this is drawn for.
  const slope = Math.atan2(rise, width / 2);
  const runner = Math.hypot(width / 2, rise);
  const t = Math.max(rise * 0.09, 0.1);
  const lashing = flatWinter({ color: W.WOOD_LIT });
  const stone = flatWinter({ color: W.STONE_DARK });

  for (let i = 0; i < BATTENS; i++) {
    const z = ((i + 1) / (BATTENS + 1) - 0.5) * depth;
    for (const s of [-1, 1] as const) {
      const bar = new THREE.Mesh(new THREE.BoxGeometry(runner, t, t * 1.6), lashing);
      bar.position.set((s * width) / 4, rise / 2, z);
      bar.rotation.z = -s * slope;
      bar.castShadow = bar.receiveShadow = true;
      group.add(bar);

      // A weight at the foot of every batten, hanging just past the eave. These put
      // BUMPS on the lower outline — the half of the silhouette a ridge cannot reach.
      const weight = new THREE.Mesh(new THREE.BoxGeometry(t * 3.4, t * 2.6, t * 3), stone);
      weight.position.set((s * width) / 2, -t, z);
      weight.castShadow = weight.receiveShadow = true;
      group.add(weight);
    }
  }
  return group;
}

/**
 * THE BANKING: earth and brush heaped against the walls, and the SECONDARY cue.
 *
 * Historically the honest answer and nearly invisible at 49°, so it is not asked to
 * carry anything. What it buys is the near wall — a dark band at the foot separates a
 * reinforced building from a bare one even when the roof is behind something. DARK
 * because everything at ground level here is near-white and the only value left is
 * downward, and OUTSIDE the wall where the drifts are, never inside it where the frost
 * crust climbs: the two must not occupy the same band.
 */
export function buildShieldBerm(footprint: Footprint): THREE.Group {
  const span = wallSpan(footprint);
  const group = new THREE.Group();
  group.name = 'winter-shield-berm';
  const earth = flatWinter({ color: W.SLUSH_DARK, roughness: 0.98 });
  const brush = flatWinter({ color: W.WOOD_DARK, roughness: 0.98 });

  const sides = [
    { rot: 0,             len: span.x, off: [0, -span.z / 2] },
    { rot: Math.PI,       len: span.x, off: [0,  span.z / 2] },
    { rot: -Math.PI / 2,  len: span.z, off: [ span.x / 2, 0] },
    { rot: Math.PI / 2,   len: span.z, off: [-span.x / 2, 0] },
  ] as const;

  for (const side of sides) {
    // A WEDGE, not a box: the bank leans on the wall, so its top edge is inboard of
    // its foot. A box would read as a plinth the building was always standing on.
    const wedge = new THREE.Shape();
    wedge.moveTo(0, 0); wedge.lineTo(BERM_REACH, 0); wedge.lineTo(0, BERM_RISE); wedge.closePath();
    const geo = new THREE.ExtrudeGeometry(wedge, { depth: side.len * 0.94, bevelEnabled: false });
    geo.translate(0, 0, (-side.len * 0.94) / 2);
    geo.rotateY(Math.PI / 2);
    const bank = new THREE.Mesh(geo, earth);
    bank.rotation.y = side.rot;
    bank.position.set(side.off[0], 0.02, side.off[1]);
    bank.castShadow = bank.receiveShadow = true;
    group.add(bank);

    for (const at of [0.3, 0.68]) {         // two leaning stakes pinning each bank
      const stake = new THREE.Mesh(new THREE.BoxGeometry(0.16, BERM_RISE * 1.5, 0.16), brush);
      stake.rotation.y = side.rot;
      stake.position.set(side.off[0], BERM_RISE * 0.55, side.off[1]);
      stake.translateX((at - 0.5) * side.len * 0.8);
      stake.translateZ(-BERM_REACH * 0.55);
      stake.rotation.x = 0.22;
      stake.castShadow = stake.receiveShadow = true;
      group.add(stake);
    }
  }
  return group;
}

// ── THE PROPOSAL IS THE SAME OBJECT, TRANSLUCENT ─────────────────────
/**
 * Not a second geometry and not a frame around the building: there is ONE shield in
 * this codebase and the ghost IS it, which is what makes the preview a promise the
 * commit is obliged to keep.
 *
 * Materials are CLONED before they are touched — Object3D.clone() shares them and the
 * cap template is shared by every building on that plot shape, so mutating in place
 * would ghost every shield already standing. Deduplicated by the ORIGINAL material's
 * identity, so the six lashings that share one material still share one ghost of it.
 */
export function ghostWinterShield(root: THREE.Object3D, opacity = GHOST_OPACITY): number {
  const ghosted = new Map<THREE.Material, THREE.Material>();
  let touched = 0;
  root.traverse((child) => {
    const mesh = child as THREE.Mesh;
    if (!mesh.isMesh) return;
    mesh.castShadow = mesh.receiveShadow = false;      // a proposal casts nothing
    const swap = (m: THREE.Material) => {
      const already = ghosted.get(m);
      if (already) return already;
      const clone = m.clone();
      clone.transparent = true;
      clone.opacity = opacity;
      clone.depthWrite = false;   // never occludes the roofline it is proposed over
      clone.needsUpdate = true;
      ghosted.set(m, clone);
      return clone;
    };
    mesh.material = Array.isArray(mesh.material) ? mesh.material.map(swap) : swap(mesh.material);
    touched++;
  });
  return touched;
}

// ── STAGING THE FIVE ─────────────────────────────────────────────────
// The cap is lifted to the LIVE measured roofline, the way MasteryStars measures for
// its star row, because these buildings are not the same height and a constant would
// float the roof on the short ones and bury it on the tall ones.
export function dressBuilding(model: THREE.Object3D, state, footprint: Footprint) {
  const top = new THREE.Box3().setFromObject(model).max.y;
  const dressing = new THREE.Group();
  dressing.name = `winter-protection-${state}`;

  if (state === 'protected' || state === 'proposed') {
    const cap = buildShieldCap(footprint);
    cap.position.y = top;                    // eaves at y = 0, so this lands them on the roof
    dressing.add(cap, buildShieldBerm(footprint));
    if (state === 'proposed') ghostWinterShield(dressing);
  }
  if (state === 'frosted') dressing.add(buildFrostCrust(footprint));   // batch 1, asset 7
  if (state === 'ruined')  dressing.add(buildRuinHeap(footprint));     // batch 1, asset 8
  // 'threatened' adds NOTHING to the building. Its whole mark is the badge, asset 5.
  return dressing;
}

The two crisis badges

Output: three.js code

Three panels: threatened, lost, and a building that is shielded and threatened at once. The badge is a flat screen-facing plate floating clear above the building's measured roofline, in the slot the mastery stars use the rest of the time, and it replaces them — the slot holds one thing legibly and two things never. There is no safe badge: an unthreatened standing building gets nothing added and keeps its stars right through the winter.

import * as THREE from 'three';
import { WINTER_GROUND_TONES, WINTER_SCENE_TONES as W } from './winterTones';

/** A runtime tuple, so a guard can sweep the population instead of restating it. */
export const CRISIS_STATUSES = ['threatened', 'lost'] as const;

/**
 * Both tones read off the winter's existing tables and NEITHER added.
 *
 * `threatened` is the ice front's own swatch on purpose: the island's legend already
 * glosses it as "the ice front: where it reaches this round", and a threatened building
 * is exactly a building the front reaches this round. One vocabulary, not two.
 */
const MARKER_TONES = {
  threatened: { face: WINTER_GROUND_TONES.frostFront, edge: W.CHAR },        // #3C7FA0 on #3A342C
  lost:       { face: W.CHAR,                         edge: W.SLUSH_DARK },  // #3A342C on #6F6357
} as const;

export const MARKER_SIZE = 2.5;      // world units
export const MARKER_MARGIN = 2.4;    // gap above the MEASURED roofline
const FALLBACK_TOP_Y = 6;            // until the model mesh is measurable

/**
 * One band of a chevron driving DOWNWARD, at `offset` up the badge.
 *
 * Down rather than up, and POINTED rather than blunt: the frost arrives from outside and
 * comes in, so the badge points the way the harm is travelling. Two stacked is the thing
 * no other mark on this island looks like.
 */
function chevronShape(offset: number): THREE.Shape {
  const w = 0.5, thickness = 0.19, drop = 0.22;
  const s = new THREE.Shape();
  s.moveTo(-w, offset + drop);
  s.lineTo(0,  offset - drop);
  s.lineTo(w,  offset + drop);
  s.lineTo(w,  offset + drop - thickness);
  s.lineTo(0,  offset - drop - thickness);
  s.lineTo(-w, offset + drop - thickness);
  s.closePath();
  return s;
}

/**
 * One arm of a BROKEN cross. Four of them, radiating diagonally, with a clear gap where
 * they would meet — `gap` is 0.09 against a 0.52 half-length, so the hole is 17% of the
 * arm and survives being seen small. A cross that CLOSES is a different mark and a
 * commoner one; the break is the whole silhouette.
 */
function brokenArmShape(rotation: number): THREE.Shape {
  const half = 0.52, thickness = 0.15, gap = 0.09;
  const corners: [number, number][] = [
    [gap, -thickness], [half, -thickness], [half, thickness], [gap, thickness],
  ];
  const s = new THREE.Shape();
  corners.forEach(([x, y], i) => {
    const rx = x * Math.cos(rotation) - y * Math.sin(rotation);
    const ry = x * Math.sin(rotation) + y * Math.cos(rotation);
    i === 0 ? s.moveTo(rx, ry) : s.lineTo(rx, ry);
  });
  s.closePath();
  return s;
}

export function crisisMarkerGeometries(status): THREE.ShapeGeometry[] {
  const shapes = status === 'threatened'
    ? [chevronShape(0.30), chevronShape(-0.08)]
    : [Math.PI / 4, (3 * Math.PI) / 4, (5 * Math.PI) / 4, (7 * Math.PI) / 4].map(brokenArmShape);
  return shapes.map((s) => new THREE.ShapeGeometry(s));
}

/**
 * The badge: a dark backing plate a touch larger than the face, then the face.
 *
 * MasteryStars' own construction, for the same reason — a flat shape against a
 * near-white overcast sky needs an outline or it dissolves into it, and that is the one
 * background every badge on this island is seen against. Both meshes are DoubleSide,
 * because the compass orbits and a one-sided badge disappears from behind.
 */
export function buildCrisisBadge(status): THREE.Group {
  const tone = MARKER_TONES[status];
  const group = new THREE.Group();
  group.name = 'crisis-status-marker';
  group.userData = { crisisStatus: status };
  for (const geometry of crisisMarkerGeometries(status)) {
    const plate = new THREE.Mesh(geometry, new THREE.MeshStandardMaterial({
      color: tone.edge, roughness: 0.8, metalness: 0, side: THREE.DoubleSide }));
    plate.scale.setScalar(MARKER_SIZE * 1.24);
    plate.position.z = -0.05;
    const face = new THREE.Mesh(geometry, new THREE.MeshStandardMaterial({
      color: tone.face, roughness: 0.55, metalness: 0, side: THREE.DoubleSide }));
    face.scale.setScalar(MARKER_SIZE);
    plate.raycast = face.raycast = () => null;   // clicks fall through to the building
    group.add(plate, face);
  }
  return group;
}

// ── THE OVERLAP, WHICH IS THE PANEL THAT MATTERS ─────────────────────
/**
 * A building can be shielded AND threatened at once, and that is the moment reinforce
 * justifies itself. It is also the frame most likely to turn to mush, so the two facts
 * are composed on channels that cannot collide:
 *
 *   the shield is the building's own GEOMETRY  — a storm roof, an outline change;
 *   the threat is a BILLBOARD ABOVE it         — screen-facing, clear of everything.
 *
 * The measure group therefore contains the model AND the shield, so the badge rides
 * above the storm roof rather than through it. Measured live, per frame, because the
 * shield may be added or spent while the badge is up.
 */
export function placeBadge(badge: THREE.Object3D, measureGroup: THREE.Object3D, camera) {
  const box = new THREE.Box3().setFromObject(measureGroup);
  const top = Number.isFinite(box.max.y) ? box.max.y : FALLBACK_TOP_Y;
  const sc-camel-parent-y = badge.parent ? badge.parent.getWorldPosition(new THREE.Vector3()).y : 0;
  badge.position.y = top - parentY + MARKER_MARGIN;
  orientStarsFacingCamera(badge, camera);       // the stars' own billboard helper
}
